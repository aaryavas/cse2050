from blockchain import Blockchain 
from blockchain import Transaction
from blockchain import Block
from blockchain import Ledger
import unittest


class TestBlockchain(unittest.TestCase):

    def test_blockchain(self):
        '''tests blockcahin to see if it empty'''
        blockchain = Blockchain()
        assert len(blockchain._blockchain) == 1
        assert blockchain.validate_chain()
        
        '''Testing a block''' 
        trans1 = Transaction("Alice", "Bob", 10)
        block1 = Block([trans1], blockchain.get_latest_block_hash())
        assert blockchain.add_block(block1)
        assert len(blockchain._blockchain) == 2
        assert blockchain.validate_chain()
        assert blockchain._bc_ledger._hashmap["Alice"] == -10
        assert blockchain._bc_ledger._hashmap["Bob"] == 10

        '''Testing an invalid block when it has incorrect previous_block_hash''' 
        transaction2 = Transaction("Charlie", "Dave", 5)
        blockchain2 = Block([transaction2], hash("invalid_hash"))
        assert not blockchain.add_block(blockchain2)
        assert len(blockchain._blockchain) == 2
        assert blockchain.validate_chain()
        assert "Charlie" not in blockchain._bc_ledger._hashmap

        '''Testing an invalid block w/ empty transactions'''
        blockchain3 = Block([])
        assert not blockchain.add_block(blockchain3)
        assert len(blockchain._blockchain) == 2
        assert blockchain.validate_chain()
        assert "Dave" not in blockchain._bc_ledger._hashmap

        '''testing for adding mutliple valid blocks'''
        transaction4 = Transaction("Alice", "Eve", 20)
        transaction5 = Transaction("Bob", "Eve", 15)
        blockchain4 = Block([transaction4, transaction5], blockchain.get_latest_block_hash())
        assert blockchain.add_block(blockchain4)
        assert len(blockchain._blockchain) == 3
        assert blockchain.validate_chain()
        assert blockchain._bc_ledger._hashmap["Eve"] == 35


        '''testing rewards to users'''
        blockchain.distribute_mining_reward("Alice")
        assert blockchain._bc_ledger._hashmap["Alice"] == 1000
        assert len(blockchain._blockchain) == 4
        assert blockchain.validate_chain()
    
        
unittest.main()
